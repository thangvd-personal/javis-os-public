**Goal:**
- T0: đưa công tắc Cộng hưởng lên từng dòng trợ lý của trang Cộng sự (một nơi duy nhất), đồng bộ hai tab, chỉ báo thành công khi server xác nhận.

**Architecture:** T0 không thêm route.

- Công tắc trùng ở cột phải và trang Cài đặt được bỏ.

**Spec:** [2026-10-13-team-chat-design.md](../specs/2026-10-13-team-chat-design.md) (D1 tới D7 đã chốt).

## Review Focus

1. **Bấm công tắc hai lần thật nhanh:** lần thứ hai không gửi khi lần đầu chưa xong (công tắc khoá trong lúc lưu).
2. **Server trả 409 (agent `missing` hay cần xác nhận):** công tắc về đúng trạng thái thật sau khi tải lại, không giữ trạng thái người dùng vừa bấm.
3. **`GET /resonance/agents` lỗi hay chậm:** danh sách trợ lý vẫn hiện; dòng không có công tắc (không đoán trạng thái).
4. **Tên trợ lý có ký tự HTML:** nhãn aria và hộp xác nhận không chèn HTML.
5. **Tab khác đổi trạng thái:** tab này cập nhật theo gói WebSocket, không cần F5.

  /* T0 nhóm chat team (spec mục 10): trạng thái Cộng hưởng của MỘT trợ lý bằng chữ, cho dòng ở cột trái trang Cộng sự. */
  function stateKey(row) {
    var r = row || {};
    if (r.status === "missing") return "resonance.t0_state_missing";
    if (r.status === "retired") return "resonance.t0_state_retired";
    return r.enabled && r.status === "active" ? "resonance.t0_state_on" : "resonance.t0_state_off";
  }

  /* Công tắc trên dòng trợ lý. Không có dữ liệu thì không vẽ gì: không đoán trạng thái. `missing` không có ô bật mà có
     nút mở bảng bên phải, nơi có hai nút xác nhận của A1. */
  function rowToggleHtml(row, opts) {
    if (!row || !row.slug) return "";
    var o = opts || {};
    var key = stateKey(row);
    var label = o.saving ? tw("resonance.t0_saving") : tw(key);
    if (row.status === "missing" || row.status === "retired") {
      return '<button type="button" class="ws-rs ws-rs-confirm" data-rs-open="' + esc(row.slug) + '">' +
        '<span class="ws-rs-text">' + esc(label) + "</span></button>";
    }
    var on = !!row.enabled && row.status === "active";
    return '<label class="ws-rs" title="' + esc(tw("resonance.t0_toggle_aria", { name: row.name || row.slug })) + '">' +
      '<span class="ws-rs-text">' + esc(label) + "</span>" +
      '<span class="toggle"><input type="checkbox" class="ws-rs-toggle" data-rs="' + esc(row.slug) + '"' +
      ' aria-label="' + esc(tw("resonance.t0_toggle_aria", { name: row.name || row.slug })) + '"' +
      (on ? " checked" : "") + (o.saving ? " disabled" : "") + "><span></span></span></label>";
  }

  /* Hỏi lại trước khi TẮT khi trợ lý còn mục tiêu mở (hợp đồng A1: dừng ở bước kế tiếp). Không có thì không hỏi. */
  function offConfirmText(row) {
    var n = Number((row || {}).goals_open || 0);
    if (!row || !row.enabled || n <= 0) return "";
    return tw("resonance.t0_off_confirm", { name: row.name || row.slug, n: n });
